Enemy.cs


using UnityEngine;
using Gloval;

public class Enemy : MonoBehaviour
{
    public enum EnemyState
    {
        IDLE,
        MOVE,
    }
    
    BoardManager bm;
    PlayerManager pm;
    GameManager gm;
    //EnemyGenerator eg;

    [Tooltip("状態変数"), SerializeField]
    EnemyState state = EnemyState.IDLE;

    [Tooltip("移動の目標地点") ,SerializeField]
    Vector3 targetPos;
    [Tooltip("移動速度")]
    float moveSpeed = 3f;
    [Tooltip("移動速度乱数")]
    float randMoveSpeedRatio;

    [Tooltip("待機時間計測")] 
    float stateTimer;

    [Tooltip("消滅アニメーション"), SerializeField]
    GameObject prfbBreakAnim;

    // Start is called before the first frame update
    void Start()
    {
        bm = GameObject.Find("BoardManager").GetComponent();
        pm = GameObject.Find("Player").      GetComponent();
        gm = GameObject.Find("GameManager"). GetComponent();

        //eg = GameObject.Find("EnemyGenerator").GetComponent();
    }

    // Update is called once per frame
    void Update()
    {
        //ゲーム中のみ.
        if (gm.startFlag && !gm.gameOverFlag)
        {
            // 状態遷移処理
            State();

            Vector2Int position = Gl_Func.WPosToBPos(transform.position);
            if (bm.Board[position.x, position.y].type == BoardType.PLAYER_AREA)
            {
                EnemyDeath();
            }
        }
    }

    /// 
    /// 状態ごとの処理
    /// 
    void State()
    {
        switch (state)
        {
            case EnemyState.IDLE:
                //待機
                EnemyIdle();
                break;
            case EnemyState.MOVE:
                //移動
                EnemyMove();
                break;
        }
    }

    /// 
    /// 待機状態処理
    /// 
    void EnemyIdle()
    {
        if(stateTimer > 0)
        {
            stateTimer -= Time.deltaTime;
        }
        else
        {
            print("待機終了");
            SetGoalMove();
        }
    }

    // todo:敵の動き
    /// 
    /// 敵の移動処理  
    /// 
    void EnemyMove()
    {
        var tmpPos = transform.position; //現在座標の仮取得.
        var dis = targetPos - tmpPos;    //目標地点までの座標差(distance)

        //仮で移動したとする.
        tmpPos += dis.normalized * moveSpeed * randMoveSpeedRatio * Time.deltaTime;
        
        //何のマスに乗ってるかで処理を分ける.
        Vector2Int bpos = Gl_Func.WPosToBPos(tmpPos);
        switch (bm.Board[bpos.x, bpos.y].type)
        {
            case BoardType.PLAYER_AREA:
                SetGoalMove();
                break;
            case BoardType.PLAYER_TRAIL:
                pm.PlayerDeath();
                break;
            case BoardType.NONE:
                transform.position = tmpPos; //実際に移動する.
                break;
        }
        
        //目標地点に着いたら(一定範囲)
        if ((targetPos - transform.position).magnitude < Gl_Const.ENM_GOAL_STOP_RANGE)
        {
            SetGoalMove();
        }
    }
    
    void EnemyDeath()
    {
        gm.deathEnemyCnt++;  //死亡数+1.
        
        var obj = Instantiate(prfbBreakAnim);
        obj.transform.position = transform.position;
        
        Destroy(gameObject); //自身を消滅.
    }

    /// 
    /// 移動開始時の処理と、目標座標の決定
    /// 
    void SetGoalMove()
    {
        print("移動開始");
        state = EnemyState.MOVE;
        
        Vector2 pos = transform.position;
        pos = Gl_Func.LimPosInBoard(pos);
        transform.position = pos;

        //目標地点の抽選.
        {
            int rndX = Random.Range(0, Gl_Const.BOARD_WID - 1);
            int rndY = Random.Range(0, Gl_Const.BOARD_HEI - 1);

            //その座標が無マスなら.
            if (bm.Board[rndX, rndY].type == BoardType.NONE)
            {
                var wPos = Gl_Func.BPosToWPos(new Vector2Int(rndX, rndY));
                targetPos = Gl_Func.LimPosInBoard(wPos);
            }
        }

        randMoveSpeedRatio = Random.Range(
            Gl_Const.ENM_MIN_MOVE_SPEED, Gl_Const.ENM_MAX_MOVE_SPEED
        );
    }
}

EnemyGenerator.cs


using System;
using System.Collections;
using UnityEngine;
using Gloval;

public class EnemyGenerator : MonoBehaviour
{
    [Tooltip("生成するプレハブ"), SerializeField]
    public GameObject prefabItem;

    [Tooltip("プレハブを入れるオブジェクト"), SerializeField]
    GameObject prefabInObj;

    [Tooltip("GameManagerのscript"), SerializeField]
    GameManager scptGameMng;

    [Tooltip("BoardManagerのscript"), SerializeField]
    BoardManager scptBoardMng;

    void Start()
    {
        StartCoroutine(WaitStart()); 
    }

    /// 
    /// スタートするまで待機する用.
    /// 
    /// 
    public IEnumerator WaitStart()
    {
        //スタートを押されるまでループ.
        while (!scptGameMng.startFlag)
        {
            yield return null;
        }

        //モード別で出現実行.
        switch (scptGameMng.gameMode) {

            case GameMode.TimeUp:
                StartCoroutine(EnmSpawnTimeUp());
                break;
        
            case GameMode.AllBreak:
                StartCoroutine(EnmSpawnAllBreak());
                break;
        }
    }

    /// 
    /// ランダム生成処理(TimeUpモード)
    /// 
    public IEnumerator EnmSpawnTimeUp()
    {
        yield return new WaitForSeconds(3); //3秒の猶予.

        //最初に何体か出す.
        for (int i = 0; i < Gl_Const.ENM_TIMEUP_INIT_CNT; i++) 
        {
            EnemySpawnExe();
            yield return new WaitForSeconds(0.1f);
        }

        //ゲーム終了するまでループ.
        while (!scptGameMng.gameOverFlag)
        {
            //最大出現数になってるなら待機.
            if (scptGameMng.GetEnemyCount() >= Gl_Const.ENM_TIMEUP_MAX_CNT)
            {
                yield return null;
                continue;
            }

            //遅延時間の抽選.
            float delay = UnityEngine.Random.Range(
                Gl_Const.ENM_TIMEUP_MIN_INTERVAL,
                Gl_Const.ENM_TIMEUP_MAX_INTERVAL
            );
            yield return new WaitForSeconds(delay);

            //敵を追加.
            EnemySpawnExe();
        }
    }

    /// 
    /// ランダム生成処理(AllBreakモード)
    /// 
    public IEnumerator EnmSpawnAllBreak()
    {
        yield return new WaitForSeconds(3); //3秒の猶予.

        //最初に何体か出す.
        for (int i = 0; i < Gl_Const.ENM_ALLBREAK_MAX_CNT; i++)
        {
            EnemySpawnExe();
            yield return new WaitForSeconds(0.1f);
        }
    }

    /// 
    /// 敵を出現.
    /// 
    public void EnemySpawnExe()
    {
        //出現座標抽選.
        int rndX = UnityEngine.Random.Range(10, Gl_Const.BOARD_WID - 1);
        int rndY = UnityEngine.Random.Range(10, Gl_Const.BOARD_HEI - 1);

        //その座標が無マスなら.
        if (scptBoardMng.Board[rndX, rndY].type == BoardType.NONE)
        {
            //敵出現.
            var obj = Instantiate(prefabItem, prefabInObj.transform);
            //座標を抽選して配置.
            obj.transform.position = Gl_Func.BPosToWPos(new Vector2Int(rndX, rndY));
        }
    }

    /// 
    /// 敵のオブジェクトを取得する処理
    /// 
    public GameObject[] GetEnemyObjects()
    {
        // タグ「Enemy」を持つ全てのオブジェクトの取得
        GameObject[] Square = GameObject.FindGameObjectsWithTag("Enemy");

        return Square;
    }
}

using System.Collections;
using System.Collections.Generic;
using UnityEngine;

public class Player : MonoBehaviour
{
    BulletManager bm;

    GameManager gm;

    [Tooltip("マウスカーソル取得")]

    [Tooltip("アイテム取得")]

    public int hp { get; private set; } = Common.PLAYER_HP;

    [Tooltip("アニメーター")]
    Animator animator;

    [Tooltip("持っているか")]
    bool isHaveItem = false;
    [Tooltip("所持アイテムID")]
    BulletType haveItemType;

    // Start is called before the first frame update
    void Start()
    {
        bm = GameObject.Find("BulletManager").GetComponent();
        ig = GameObject.Find("ItemGenerator").GetComponent();
        gm = GameObject.Find("GameManager").GetComponent();

        animator = GetComponent();
    }

    // Update is called once per frame
    void Update()
    {
        if (gm.isStarted)
        {
            PlayerMovement();
            NotOutsideScreen();
            ThrowButton();
            //if(Input.GetKeyDown(KeyCode.Space)) {
            //    RemainingLives();
            //}
        }

    }

    void OnTriggerEnter2D(Collider2D c)
    {

        var tag = c.gameObject.tag;
        if (tag == "Item")
        {
            if (isHaveItem != true)
            {
                haveItemType = c.gameObject.GetComponent().bulletType;
                gm.GetItem(haveItemType);
                Destroy(c.gameObject);
                isHaveItem = true;
                ig.DecCnt();
            }
        }


    }

    // todo.上下左右に移動
    /// 
    /// プレイヤー移動処理
    /// 
    public void PlayerMovement()
    {
        //var(lb, rt) = Common.GetWorldWindowSize();
        bool isMove = false;

        if ((Input.GetKey(KeyCode.UpArrow)) || (Input.GetKey(KeyCode.W)))
        {
            transform.position += new Vector3(0, 1, 0) * moveSpeed * Time.deltaTime;
            isMove = true;
        }
        if ((Input.GetKey(KeyCode.DownArrow)) || (Input.GetKey(KeyCode.S)))
        {
            transform.position += new Vector3(0, -1, 0) * moveSpeed * Time.deltaTime;
            isMove = true;
        }
        if ((Input.GetKey(KeyCode.LeftArrow)) || (Input.GetKey(KeyCode.A)))
        {
            transform.position += new Vector3(-1, 0, 0) * moveSpeed * Time.deltaTime;
            var ls = transform.localScale;
            if(ls.x < 0)
            {
                ls.x *= -1;
            }
            transform.localScale = ls;
            isMove = true;
        }
        if ((Input.GetKey(KeyCode.RightArrow)) || (Input.GetKey(KeyCode.D)))
        {
            transform.position += new Vector3(1, 0, 0) * moveSpeed * Time.deltaTime;
            var ls = transform.localScale;
            if (ls.x > 0)
            {
                ls.x *= -1;
            }
            transform.localScale = ls;
            isMove = true;
        }

        if (isMove)
        {
            animator.ResetTrigger("Idle");
            animator.SetTrigger("Move");
        }
        else
        {
            animator.ResetTrigger("Move");
            animator.SetTrigger("Idle");
        }
    }

    // todo.画面外に出ない
    /// 
    /// プレイヤーが画面外に出ない処理
    /// 
    public void NotOutsideScreen()
    {
        var (lb, rt) = Common.GetWorldWindowSize();

        if (transform.position.x < lb.x + Common.MARGIN_LEFT)
        {
            transform.position = new Vector3(lb.x + Common.MARGIN_LEFT, transform.position.y, transform.position.z);
        }
        if (transform.position.x > 0 - Common.MARGIN_RIGHT)
        {
            transform.position = new Vector3(0 - Common.MARGIN_RIGHT, transform.position.y, transform.position.z);
        }
        if (transform.position.y > rt.y - Common.MARGIN_TOP)
        {
            transform.position = new Vector3(transform.position.x, rt.y - Common.MARGIN_TOP, transform.position.z);
        }
        if (transform.position.y < lb.y + Common.MARGIN_BOTTOM)
        {
            transform.position = new Vector3(transform.position.x, lb.y + Common.MARGIN_BOTTOM, transform.position.z);
        }

    }

using System.Collections;
using System.Collections.Generic;
using Unity.VisualScripting;
using UnityEngine;
using static UnityEditor.Experimental.GraphView.GraphView;
using static UnityEngine.GraphicsBuffer;

public class ObjectLocation : MonoBehaviour
{
    #region 宣言

    [SerializeField] private int weight; // 重さ
    [SerializeField] private int power;  // このオブジェクトにかかっている力

    public GameObject player1;
    public GameObject player2;

    Player pl1;
    Player pl2;

    [SerializeField] int holdCnt = 0;

    [SerializeField] bool isGrab1 = false;
    [SerializeField] bool isGrab2 = false;
    public bool isCheck = false;
    bool isStopping = false;

    float distancePlayer;
    float baseAngle; // 初期角度

    public Vector3 hitNormal;
    Vector2 prevPos1;
    Vector2 prevPos2;
    Rigidbody2D rb;
    #endregion

    #region Set関数
    public void AddPower(int _power)
    {
        power += _power;
    }

    public void SubtractPower(int _power)
    {
        power -= _power;
        if (power < 0) { power = 0; }
    }

    public void SetIsGrab1(bool flg)
    {
        isGrab1 = flg;
    }

    public void SetIsGrab2(bool flg)
    {
        isGrab2 = flg;
    }

    public void AddHoldCnt()
    {
        holdCnt++;
    }

    public void SubtractHoldCnt()
    {
        holdCnt--;
    }
    #endregion

    #region Get関数
    public bool GetCanHold()
    {
        return power >= weight;
    }

    public int GetHoldCnt()
    {
        return holdCnt;
    }

    public bool GetIsGrab1()
    {
        return isGrab1;
    }

    public bool GetIsGrab2()
    {
        return isGrab2;
    }
    #endregion

    #region 回転処理

    /// 
    /// 家具を回転させる処理
    /// 
    void FurnitureLocation()
    {
        Vector2 posA = player1.transform.position;
        Vector2 posB = player2.transform.position;

        // 中間位置
        Vector2 center = (posA + posB) / 2f;
        rb.MovePosition(center);

        Vector2 leftPos, rightPos;

        if(posA.x <= posB.x)
        {
            leftPos = posA;
            rightPos = posB;
        }
        else
        {
            leftPos = posB;
            rightPos = posA;
        }

        // 向き
        Vector2 dir = rightPos - leftPos;
        float angle = Mathf.Atan2(dir.y, dir.x) * Mathf.Rad2Deg;
        rb.MoveRotation(angle);
    }

    void AdjustPlayers()
    {
        Vector2 inputA = pl1.GetMoveInput(); // 入力を取る
        Vector2 inputB = pl2.GetMoveInput();

        // 微小入力カット
        if (inputA.magnitude < 0.1f) inputA = Vector2.zero;
        if (inputB.magnitude < 0.1f) inputB = Vector2.zero;

        Vector2 posA = player1.transform.position;
        Vector2 posB = player2.transform.position;
        Vector2 dir = (posB - posA).normalized;
        Vector2 moveA = posA - prevPos1;
        Vector2 moveB = posB - prevPos2;
        float magA = moveA.magnitude;
        float magB = moveB.magnitude;

        // 両方入力してる
        if (inputA != Vector2.zero && inputB != Vector2.zero)
        {
            float dot = Vector2.Dot(inputA.normalized, inputB.normalized);

            if (dot > 0.5f) // 同じ方向
            {
                Vector2 move = (inputA + inputB) / 2f * Time.deltaTime * 5f;

                player1.transform.position += (Vector3)move;
                player2.transform.position += (Vector3)move;
            }
            else
            {
                // 方向違うなら止める
                player1.transform.position = prevPos1;
                player2.transform.position = prevPos2;
            }
        }
        //プレイヤー1が止まっている時プレイヤー2だけ動く
            else if (magA < 0.01f && magB > 0.01f)
        {
            if (isStopping) return;
            print("D");
            Vector2 newPosB = posA + dir * distancePlayer;
            player2.transform.position = newPosB;
        }
        // プレイヤー2が止まっているときにプレイヤー1だけ動く
        else if (magB < 0.01f && magA > 0.01f)
        {
            if (isStopping) return;

            print("E");
            Vector2 newPosA = posB - dir * distancePlayer;
            player1.transform.position = newPosA;
        }
    }

    /// 
    /// 運ぶ処理
    /// 
    /// 
    public void Group(GameObject player)
    {
        if (player == player1) isGrab1 = true;
        if (player == player2) isGrab2 = true;

        if (isGrab1 && isGrab2)
        {
            //rb.isKinematic = true;
            if (pl1.GetHoldFlg()) pl1.SetHoldFlg(false);
            if (pl2.GetHoldFlg()) pl2.SetHoldFlg(false);

            rb.constraints = RigidbodyConstraints2D.None;

            // 距離保存
            distancePlayer = Vector2.Distance(player1.transform.position, player2.transform.position);

            // 初期角度を保存
            //Vector2 dir = player2.transform.position - player1.transform.position;
            //baseAngle = Mathf.Atan2(dir.y, dir.x) * Mathf.Rad2Deg;
        }
        else if(!isGrab1 || !isGrab2)
        {
            rb.constraints = RigidbodyConstraints2D.FreezeAll;
        }
    }
    #endregion

    #region Unityイベント
    // Start is called before the first frame update
    void Start()
    {
        rb = GetComponent();
        pl1 = player1.GetComponent();
        pl2 = player2.GetComponent();

        // rb.isKinematic = true;
    }

    // Update is called once per frame
    void Update()
    {
        if (isGrab1 && isGrab2)
        {
            FurnitureLocation();
            AdjustPlayers();
        }
    }

    void FixedUpdate()
    {
        prevPos1 = player1.transform.position;
        prevPos2 = player2.transform.position;
    }

        if (col.gameObject.CompareTag("Obstacle") || col.gameObject.CompareTag("Furniture"))

ObjectLife.cs(オブジェクトが壁に当たった時に体力を減らす処理)



using System.Collections;
using System.Collections.Generic;
using Unity.VisualScripting;
using UnityEngine;

public class ObjectLife : MonoBehaviour
{
    #region 宣言
    [SerializeField]
    int scoreObj = 100;

    bool isTemetutyu = false;
    bool isStop = false;
    bool isBlink = false;

    float cooldown = 3;

    SpriteRenderer sr;
    KoguchiUI koguchiUI;
    [SerializeField]Clear clear;
    
    #endregion

    #region 家具のスコアを減らす処理

    /// 
    /// スコアを削減する処理
    /// 
    public void ScoreLower(int score)
    {
        scoreObj -= score;
        isTemetutyu = true;
        OnDamege();
        
    }

    void OnDamege()
    {
        InvokeRepeating("Blink", 0f, 0.5f);
        Invoke("StopBlink", cooldown);
    }

    void Blink()
    {
        isBlink = !isBlink;
        sr.enabled = isBlink;
    }

    void StopBlink()
    {
        CancelInvoke("Blink"); // 点滅停止
        sr.enabled = true; // 最後は表示に戻す
        isTemetutyu = false;
    }

    #endregion

    #region Unityイベント
    // Start is called before the first frame update
    void Start()
    {
       sr = GetComponent();
       koguchiUI = GameObject.Find("KoguchiUI").GetComponent();
       clear = GameObject.Find("ClearManager").GetComponent();
    }

    // Update is called once per frame
    void Update()
    {
        if (isTemetutyu) return;
        
        if (isStop)
        {
            ScoreLower(10);
        }
    }

    void OnCollisionEnter2D(Collision2D col)
    {
        if (col.gameObject.CompareTag("Furniture"))
        {
            isStop = true;
        }
    }

    void OnCollisionExit2D(Collision2D collision)
    {
        if (collision.gameObject.CompareTag("Furniture"))
        {
            isStop = false;
        }
    }

    void OnTriggerEnter2D(Collider2D collision)
    {
        print("あたった");
        if (collision.gameObject.CompareTag("CollectArea"))
        {
            koguchiUI.AddScore(scoreObj);
            clear.ClearTotal(1);
            Destroy(gameObject);
        }
    }
    #endregion
}
